import type { CSSProperties } from 'react'

type Step = { h: number; hm: number; t: number; rs: number; ys: number }

const STEPS: Step[] = [
  { h: 168, hm: 148, t: 7, rs: 8, ys: 11 },
  { h: 144, hm: 136, t: 6, rs: 7, ys: 10 },
  { h: 120, hm: 124, t: 5, rs: 6, ys: 9 },
  { h: 108, hm: 116, t: 4, rs: 5, ys: 8 },
  { h: 96, hm: 108, t: 4, rs: 5, ys: 8 },
  { h: 84, hm: 100, t: 4, rs: 5, ys: 8 },
  { h: 72, hm: 92, t: 4, rs: 5, ys: 8 },
]

const LAST: Step = { h: 72, hm: 92, t: 4, rs: 5, ys: 8 }

// Row heights step down as the list runs on. Values travel as custom properties
// so the static classes in each row can read them.
export function stepStyle(index: number): CSSProperties {
  const step = STEPS[Math.max(0, Math.min(index, STEPS.length - 1))] ?? LAST
  return {
    '--h': `${step.h}px`,
    '--hm': `${step.hm}px`,
    '--t': String(step.t),
    '--rs': String(step.rs),
    '--ys': String(step.ys),
  } as CSSProperties
}

function featuredNote(names: string[]): string {
  if (names.length === 0) return 'Featured.'
  const shown = names.slice(0, 3)
  const last = shown[shown.length - 1] ?? ''

const ordered = [...(featuredProject ? [featuredProject] : []), ...selectedWork, ...experiments]

function neighbours(slug: string): number[] {
  const at = ordered.findIndex((p) => p.slug === slug)
  if (at < 0) return []
  return [at - 1, at + 1].filter((n) => n >= 0 && n < ordered.length)
}
