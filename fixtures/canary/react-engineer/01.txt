// Ranges print with an en dash and no spaces. An open range ("2025 —") prints as "2025 to present",
// so no separator is left dangling.
function formatYear(raw: string): string {
  const trimmed = raw.trim()
  const open = /[\u2014\u2013-]\s*$/.test(trimmed)
  const base = trimmed.replace(/\s*[\u2014\u2013-]\s*$/, '')
  if (open) return `${base} to present`
  return base.replace(/\s*[\u2014\u2013]\s*/g, '\u2013')
}

function Line({ text, company }: { text: string; company?: boolean }) {
  if (!text) return null
  return <p className={company ? companyClass : descClass}>{text}</p>
}

function TimelineRow(props: {
  year: string
  role: string
  company: string
  description: string
  step: number
}) {
  return (
    <div className={row} style={stepStyle(props.step)}>
      <p className={year}>{formatYear(props.year)}</p>
      <div className={cellClass}>
        <h3 className={headClass}>{props.role || props.company}</h3>
        <Line text={props.role ? props.company : ''} company />
        <Line text={props.description} />
      </div>
    </div>
  )
}

export function AboutTimeline() {

              step={i}
            />
          </li>
        ))}
      </ol>
    </section>
  )
}
